A.1 ¿Qué es un lenguaje de guiones de servidor?

Un lenguaje de guiones de servidor se ejecuta en el servidor web y genera dinámicamente la respuesta HTTP (normalmente HTML) que se envía al cliente. El código original nunca llega al navegador: solo el resultado.

javascript

  const nombre = req.query.nombre || 'invitado';

from flask import Flask, request

Código en el servidor (saludo.php):

?>

  <h1>Hola, <?= htmlspecialchars($nombre) ?> 👋</h1>

Lo que recibe el navegador (Ver código fuente):

html 
<!DOCTYPE html>
<html>

<body>
  <h1>Hola, Ana 👋</h1>
  <p>Fecha: 14/03/2026 10:23</p>
</body>
</html>
🔒
Nunca aparece el <?php ... ?>.

Si aparece, el servidor no está interpretando PHP.

B.3 Generación dinámica de CSS y JS

El servidor también puede generar CSS o JS dinámicamente:

php 
/* estilos.php */
<?php
header('Content-Type: text/css');
$color = $_COOKIE['tema'] ?? 'light';
?>
body {
  background: <?= $color === 'dark' ? '#222' : '#fff' ?>;
  color: <?= $color === 'dark' ? '#eee' : '#000' ?>;
}
html 
<link rel="stylesheet" href="estilos.php">

Enunciado: Crea dos funciones:

  • calcularIMC(float $peso, float $altura): float — devuelve IMC redondeado a 2 decimales.
  • categoriaIMC(float $imc): string — devuelve "Bajo peso" (<18.5), "Normal" (18.5–24.9), "Sobrepeso" (25–29.9) u "Obesidad" (≥30).

<?php
declare(strict_types=1);

function calcularIMC(float $peso, float $altura): float {
    if ($altura <= 0) {
        throw new InvalidArgumentException("Altura inválida");
    }
    return round($peso / ($altura ** 2), 2);
}

function categoriaIMC(float $imc): string {
    return match(true) {
        $imc < 18.5 => "Bajo peso",
        $imc < 25   => "Normal",
        $imc < 30   => "Sobrepeso",
        default     => "Obesidad",
    };
}

$casos = [[50, 1.75], [70, 1.75], [95, 1.75]];
foreach ($casos as [$p, $a]) {
    $imc = calcularIMC($p, $a);
    echo "Peso: $p kg, Altura: $a m → IMC: $imc (" . categoriaIMC($imc) . ")<br>";
}

php 
<?php
// procesar.php
declare(strict_types=1);

if ($_SERVER["REQUEST_METHOD"] !== "POST") {
    http_response_code(405);
    exit("Método no permitido");
}

function e(string $s): string {
    return htmlspecialchars($s, ENT_QUOTES, 'UTF-8');
}

$nombre = trim($_POST["nombre"] ?? "");
$email  = filter_input(INPUT_POST, "email", FILTER_VALIDATE_EMAIL);
$edad   = filter_input(INPUT_POST, "edad", FILTER_VALIDATE_INT);

$errores = [];
if ($nombre === "")                    $errores[] = "El nombre es obligatorio.";
if (mb_strlen($nombre) > 50)           $errores[] = "El nombre es demasiado largo.";
if (!$email)                           $errores[] = "Email no válido.";
if ($edad === false || $edad < 0 || $edad > 120) $errores[] = "Edad incorrecta.";

if ($errores) {
    foreach ($errores as $e) echo "<p>⚠️ " . e($e) . "</p>";
} else {
    echo "<p>✅ Registrado: " . e($nombre) . " (" . e($email) . "), $edad años</p>";
}

F.4 Protección CSRF

El Cross-Site Request Forgery ocurre cuando un sitio externo envía un formulario en nombre del usuario autenticado.

session_start();
if (empty($_SESSION['csrf'])) {
    $_SESSION['csrf'] = bin2hex(random_bytes(32));
}
?>
<form method="POST" action="procesar.php">
  <input type="hidden" name="csrf" value="<?= $_SESSION['csrf'] ?>">

// procesar.php
session_start();
if (!hash_equals($_SESSION['csrf'] ?? '', $_POST['csrf'] ?? '')) {
    http_response_code(403);
    exit("CSRF token inválido");
}
// ... procesar

$finfo = finfo_open(FILEINFO_MIME_TYPE);
$tipo  = finfo_file($finfo, $_FILES['archivo']['tmp_name']);
if (!in_array($tipo, $tipos, true)) exit("Tipo no permitido");

$nombre = bin2hex(random_bytes(16)) . '.' . pathinfo($_FILES['archivo']['name'], PATHINFO_EXTENSION);
move_uploaded_file($_FILES['archivo']['tmp_name'], __DIR__ . "/uploads/$nombre");
echo "Subido: $nombre";
🔒
Nunca confíes en $_FILES['archivo']['type']

Se puede falsear. Usa finfo.

G.4 Persistencia en ficheros JSON

php 
function leerJson(string $ruta): array {
    if (!file_exists($ruta)) return [];
    $contenido = file_get_contents($ruta);
    return json_decode($contenido, true) ?? [];
}

function guardarJson(string $ruta, array $datos): void {
    file_put_contents(
        $ruta,
        json_encode($datos, JSON_PRETTY_PRINT | JSON_UNESCAPED_UNICODE),
        LOCK_EX
    );
}

$tareas = leerJson("tareas.json");
$tareas[] = ["id" => uniqid(), "texto" => "Estudiar PHP", "hecha" => false];
guardarJson("tareas.json", $tareas);
⚠️

H.2 Registro con hash

php 
// registro.php (fragmento)
declare(strict_types=1);

$usuario  = trim($_POST["usuario"] ?? "");

$password2 = $_POST["password2"] ?? "";

$errores = [];
if (!preg_match('/^[a-zA-Z0-9_]{3,20}$/', $usuario)) {
    $errores[] = "Usuario inválido (3-20, alfanumérico)";
}
if (strlen($password) < 8) {
    $errores[] = "Contraseña mínimo 8 caracteres";
}

    $errores[] = "Las contraseñas no coinciden";
}

    $usuarios = leerJson("usuarios.json");
    if (isset($usuarios[$usuario])) {
        $errores[] = "El usuario ya existe";
    } else {
        $usuarios[$usuario] = [
            "hash" => password_hash($password, PASSWORD_DEFAULT),
            "rol"  => "user",
            "creado" => date('c'),
        ];
        guardarJson("usuarios.json", $usuarios);
        echo "Usuario registrado ✅";
    }
}

H.3 Login con bloqueo por intentos

php 
// login.php
declare(strict_types=1);
session_start();

$usuario  = $_POST["usuario"] ?? "";
$password = $_POST["password"] ?? "";

$intentos = leerJson("intentos.json");
$ahora    = time();

// Limpiar intentos antiguos (>15 min)
$intentos = array_filter($intentos, fn($t) => $t > $ahora - 900);

$clave = "ip_" . ($_SERVER['REMOTE_ADDR'] ?? 'desconocida');
if (($intentos[$clave] ?? 0) >= 5) {
    exit("Demasiados intentos. Espera 15 minutos.");
}

$usuarios = leerJson("usuarios.json");

if (isset($usuarios[$usuario]) &&
    password_verify($password, $usuarios[$usuario]["hash"])) {

    session_regenerate_id(true);
    unset($intentos[$clave]);
    guardarJson("intentos.json", $intentos);

    $_SESSION["usuario"] = $usuario;
    $_SESSION["rol"]     = $usuarios[$usuario]["rol"];
    header("Location: panel.php");
    exit;
}

$intentos[$clave] = ($intentos[$clave] ?? 0) + 1;
guardarJson("intentos.json", $intentos);
echo "Credenciales incorrectas";

H.5 Roles y permisos granulares

php 
// permisos.php
const PERMISOS = [
    'invitado' => ['ver_inicio', 'ver_login'],
    'user'     => ['ver_inicio', 'ver_panel', 'editar_perfil', 'ver_tareas'],
    'admin'    => ['ver_inicio', 'ver_panel', 'editar_perfil', 'ver_tareas',
                   'gestionar_usuarios', 'ver_logs', 'borrar_cualquiera'],
];

function tienePermiso(string $rol, string $accion): bool {
    return in_array($accion, PERMISOS[$rol] ?? [], true);
}

function requerirPermiso(string $accion): void {
    session_start();
    $rol = $_SESSION['rol'] ?? 'invitado';
    if (!tienePermiso($rol, $accion)) {
        http_response_code(403);
        exit("⛔ Acceso denegado");
    }
}

Uso:

php 
// admin.php
require 'permisos.php';
requerirPermiso('gestionar_usuarios');
// ... contenido admin
